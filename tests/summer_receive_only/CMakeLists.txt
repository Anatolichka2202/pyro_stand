cmake_minimum_required(VERSION 3.21)
project(summer_receive_only LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Widgets Network SerialPort)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(SUMMER "${CMAKE_CURRENT_BINARY_DIR}/reference")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/prepare.py"
    "${REPO}" "${SUMMER}" COMMAND_ERROR_IS_FATAL ANY)
add_executable(summer_receive_only main.cpp "${SUMMER}/stand.cpp"
    "${SUMMER}/mainwindow.cpp" "${SUMMER}/timeline_widget.cpp" "${SUMMER}/session_logger.cpp")
target_include_directories(summer_receive_only PRIVATE "${SUMMER}")
target_link_libraries(summer_receive_only PRIVATE Qt6::Core Qt6::Widgets Qt6::Network Qt6::SerialPort)
if(WIN32)
    target_compile_definitions(summer_receive_only PRIVATE PYRO_WINDOWS)
else()
    target_compile_definitions(summer_receive_only PRIVATE PYRO_LINUX)
endif()
enable_testing()
add_test(NAME summer_passive COMMAND summer_receive_only --self-test)
set_tests_properties(summer_passive PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 15)
